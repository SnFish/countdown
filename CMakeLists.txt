cmake_minimum_required(VERSION 3.20)
project(countdown VERSION 0.1.1 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "Countdown requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(countdown WIN32 src/main.cpp src/countdown.rc)
target_include_directories(countdown PRIVATE src)
target_link_libraries(countdown PRIVATE user32 gdi32 comctl32 comdlg32 shell32 ole32 sapi uuid)
if(MSVC)
    target_compile_options(countdown PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/O1;/GL>")
    target_link_options(countdown PRIVATE /LTCG /OPT:REF /OPT:ICF /MANIFEST:NO)
elseif(MINGW)
    target_compile_options(countdown PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Os;-flto;-fno-exceptions;-fno-rtti;-ffunction-sections;-fdata-sections;-Wall;-Wextra;-Wno-missing-field-initializers>")
    target_link_options(countdown PRIVATE -municode -flto -static-libgcc -static-libstdc++ -Wl,--gc-sections,--strip-all)
else()
    message(FATAL_ERROR "Use MSVC or MinGW-w64 to build Countdown.")
endif()
add_custom_command(TARGET countdown POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_SOURCE_DIR}/dist"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:countdown>" "${CMAKE_CURRENT_SOURCE_DIR}/dist/countdown.exe"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/dist/LICENSE"
    VERBATIM)
include(CTest)
if(BUILD_TESTING)
    add_executable(timer_test tests/timer_test.cpp)
    target_include_directories(timer_test PRIVATE src)
    add_test(NAME timer COMMAND timer_test)
endif()
